\begin{frame}{What We Built}
  \begin{itemize}
    \item Automatic host discovery from \texttt{https://tp.telecom-paris.fr/ajax.php}
    \item Go orchestrator that builds, deploys, starts, monitors, and cleans workers
    \item HTTP worker API for \texttt{load -> map -> shuffle -> reduce -> result}
    \item Direct Common Crawl WET download from \texttt{data.commoncrawl.org}
    \item Four jobs: \texttt{wordcount}, \texttt{langdetect}, \texttt{domainpop}, \texttt{docdensity}
  \end{itemize}

\end{frame}

\begin{frame}[fragile]{Protocol and Dataflow}
  \begin{enumerate}
    \item \textbf{Load:} Send $\le$64MB chunks or WET URLs.
    \item \textbf{Map:} Write \texttt{N} bucket files locally.
    \item \textbf{Shuffle+reduce:} Fetch peer buckets via HTTP.
    \item \textbf{Collect:} Merge \& write result locally.
  \end{enumerate}
  \vspace{1em}
  \textbf{Map assignment:} Orchestrator split + \texttt{POST /data}\\
  \textbf{Reducer assignment:} \texttt{FNV-32a(key) \% N}
\end{frame}

\begin{frame}{Fault Tolerance and Operations}
  \textbf{Failure Detection:}
  \begin{itemize}
    \item A background health watcher polls \texttt{GET /health} every 5 seconds.
    \item Two consecutive failures mark a host as \textbf{dead}.
  \end{itemize}
  \vspace{0.5em}
  
  \textbf{Recovery Execution (What happens when a node fails):}
  \begin{itemize}
    \item The orchestrator pulls a \textbf{spare host} from the \texttt{hosts.txt} pool.
    \item It deploys a new worker binary over SSH to the spare.
    \item The system \textbf{replays} the \texttt{/load} and \texttt{/map} phases for the failed slot.
    \item If no spare host is available, the failed slot is reassigned to an existing healthy worker.
    \item Logical \textbf{slots} (\texttt{0..N-1}) stay fixed even if the physical host changes.
  \end{itemize}
\end{frame}

  \begin{table}

    \begin{tabular}{l p{0.26\linewidth} p{0.26\linewidth} p{0.26\linewidth}}
      \toprule
       & \textbf{Our MapReduce} & \textbf{Hadoop} & \textbf{Kafka Streams} \\

      Storage & \texttt{/tmp}, no replication & HDFS 3$\times$ replication & Replicated broker log \\

  \end{table}
  \begin{block}{Biggest gaps vs.\ Hadoop}
    (1) HDFS data locality — we transfer all input at load time.
    (2) Speculative execution — one slow node blocks the whole phase.
  \end{block}
  \note{Timing: 55 seconds. Kafka difference is model-level (stream vs batch). Hadoop difference is engineering-level (production data locality + scheduling we did not build).}

\begin{frame}{Pain Points: What Broke and Why?}
  During development and testing at scale, we encountered major blockers:
  \vspace{1em}
  
  \begin{enumerate}
    \vspace{0.5em}
    \item \textbf{Zombie Processes:} Killed orchestrators left orphan worker processes running on remote lab machines, locking up assigned ports.
    \begin{itemize}
      \item \textit{Fix:} Made cleanup idempotent (matching port/process ID) and added health monitoring.
    \end{itemize}
  \end{enumerate}
\end{frame}

\begin{frame}{Optimization and Future Work}
  \textbf{Performance Optimization \& Profiling}
  \begin{itemize}
    \item \textbf{In-Mapper Combiner:} Introduce an aggregation tree or combiner step in the mapper to drastically reduce HTTP requests and intermediate data size.
    \item \textbf{Streaming IO:} Stream intermediate bucket results instead of keeping maps fully in memory. This prevents OOM crashes observed with large Common Crawl files at N=1 and N=2.
  \end{itemize}
  \vspace{0.5em}
  
  \textbf{Fault Tolerance Improvements}
  \begin{itemize}
    \item \textbf{Durable State:} The current MapReduce state is volatile in \texttt{/tmp}. Implementing replicated bucket storage could provide stronger, Hadoop-like fault tolerance without full phase replays.
  \end{itemize}
\end{frame}
